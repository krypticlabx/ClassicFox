local AntiGuardEnabled = false
local IsAntiHitRunning = false
local PROTECT_DURATION = 0.35

local RoutePoints = {
    CFrame.new(500.62, 241.28, -366.64),
    CFrame.new(504.45, 155.80, -366.35),
    CFrame.new(508.30,  70.28, -366.03),
    CFrame.new(513.86,  70.28, -366.25),
    CFrame.new(519.43,  70.28, -366.47),
    CFrame.new(524.32,  70.28, -366.59),
    CFrame.new(529.22,  70.28, -366.71),
    CFrame.new(538.01,  70.28, -365.55),
    CFrame.new(546.80,  70.28, -364.40),
}
local SAFE_CF = RoutePoints[#RoutePoints]

local function getRoot()
    local char = Player.Character
    return char and char:FindFirstChild("HumanoidRootPart")
end

local function zeroVelocity(root)
    pcall(function()
        root.AssemblyLinearVelocity  = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
    end)
end

local function fireRoute()
    local root = getRoot()
    if not root then IsAntiHitRunning = false; return end

    for _, cf in ipairs(RoutePoints) do
        root.CFrame = cf
    end
    zeroVelocity(root)

    local deadline = tick() + PROTECT_DURATION
    local conn
    conn = RunService.Heartbeat:Connect(function()
        -- Kung pinatay ang Anti Guard o lumipas na ang duration, i-disconnect na
        if not AntiGuardEnabled or tick() > deadline then
            conn:Disconnect()
            IsAntiHitRunning = false
            return
        end
        local r = getRoot()
        if not r then conn:Disconnect(); IsAntiHitRunning = false; return end
        r.CFrame = SAFE_CF
        zeroVelocity(r)
    end)
end

ProximityPromptService.PromptTriggered:Connect(function(_, player)
    if player ~= Player            then return end
    if not AntiGuardEnabled        then return end -- Gagana lang kapag naka-On ang Anti Guard
    if IsAntiHitRunning            then return end
    IsAntiHitRunning = true
    task.spawn(fireRoute)
end)

local function SetAntiGuard(State)
	AntiGuardEnabled = State

	if State then
		TweenService:Create(
			Toggle,
			TweenInfo.new(0.16, Enum.EasingStyle.Quad),
			{BackgroundColor3 = Color3.fromRGB(0,145,255)}
		):Play()

		TweenService:Create(
			ToggleStroke,
			TweenInfo.new(0.16),
			{Color = Color3.fromRGB(0,200,255), Thickness = 1.5}
		):Play()

		TweenService:Create(
			Knob,
			TweenInfo.new(0.16, Enum.EasingStyle.Quad),
			{Position = UDim2.new(1,-25,0,3.5)}
		):Play()
	else
		TweenService:Create(
			Toggle,
			TweenInfo.new(0.16, Enum.EasingStyle.Quad),
			{BackgroundColor3 = Color3.fromRGB(25,30,42)}
		):Play()

		TweenService:Create(
			ToggleStroke,
			TweenInfo.new(0.16),
			{Color = Color3.fromRGB(40,50,70), Thickness = 1}
		):Play()

		TweenService:Create(
			Knob,
			TweenInfo.new(0.16, Enum.EasingStyle.Quad),
			{Position = UDim2.fromOffset(4,3.5)}
		):Play()
	end

	print("Anti Guard:", AntiGuardEnabled)
end

Toggle.MouseButton1Click:Connect(function()
	SetAntiGuard(not AntiGuardEnabled)
end)

--==================================================
-- DRAGGING
--==================================================

local Dragging = false
local DragStart
local StartPos

Header.InputBegan:Connect(function(Input)
	if Input.UserInputType == Enum.UserInputType.MouseButton1
		or Input.UserInputType == Enum.UserInputType.Touch then

		Dragging = true
		DragStart = Input.Position
		StartPos = Main.Position

		Input.Changed:Connect(function()
			if Input.UserInputState == Enum.UserInputState.End then
				Dragging = false
			end
		end)
	end
end)

UIS.InputChanged:Connect(function(Input)
	if not Dragging then return end

	if Input.UserInputType == Enum.UserInputType.MouseMovement
		or Input.UserInputType == Enum.UserInputType.Touch then

		local Delta = Input.Position - DragStart

		Main.Position = UDim2.new(
			StartPos.X.Scale,
			StartPos.X.Offset + Delta.X,
			StartPos.Y.Scale,
			StartPos.Y.Offset + Delta.Y
		)
	end
end)

local Camera = workspace.CurrentCamera

local Scale = Instance.new("UIScale")
Scale.Name = "MobileScale"
Scale.Parent = Main

local function UpdateScale()
	if not Camera then return end
	local Viewport = Camera.ViewportSize
	local S = math.min(Viewport.X / 400, Viewport.Y / 250)
	Scale.Scale = math.clamp(S, 0.70, 1)
end

UpdateScale()
Camera:GetPropertyChangedSignal("ViewportSize"):Connect(UpdateScale)
